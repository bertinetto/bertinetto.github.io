{{/* Renders an author list. Splits onto two balanced lines when len > 6. */}} {{- $websiteAuthor := .Site.Params.websiteAuthor -}} {{- $authors := .authors -}} {{- $len := len $authors -}} {{- $splitAt := 0 -}} {{- if gt $len 6 -}} {{- $splitAt = div (add $len 1) 2 -}} {{- end -}} {{- range $index, $author := $authors -}} {{- $cleanAuthor := replaceRE `\s*\*$` "" $author -}} {{- if eq $cleanAuthor $websiteAuthor -}} {{ $cleanAuthor }}{{- if strings.HasSuffix $author "*" -}}*{{- end -}} {{- else -}} {{ $cleanAuthor }}{{- if strings.HasSuffix $author "*" -}}*{{- end -}} {{- end -}} {{- if and (gt $splitAt 0) (eq (add $index 1) $splitAt) -}} {{ ", " }}
{{- else if not (eq (add $index 1) $len) -}} {{- if eq (add $index 2) $len -}}{{ " and " }}{{- else -}}{{ ", " }}{{- end -}} {{- end -}} {{- end -}}
